# Só as páginas públicas devem ser indexadas. Todo o resto exige login
# e não tem por que aparecer em busca.
#
# ATENÇÃO: "Disallow" impede a VISITA do robô, não a indexação da URL —
# um link externo ainda pode fazer a URL aparecer no resultado (sem
# conteúdo). O bloqueio de verdade é o cabeçalho X-Robots-Tag: noindex,
# configurado por caminho em vercel.json / no seu servidor. Os dois se
# complementam.

User-agent: *

# Áreas autenticadas
Disallow: /estudante
Disallow: /revisor
Disallow: /admin
Disallow: /co-chairs

# /congresso inteiro: a área do evento saiu do escopo do projeto e hoje
# exige login de admin — inclusive as páginas que antes eram públicas
# (informações, programação, verificação de certificado).
Disallow: /congresso

# Verificação de e-mail: /confirmar-email é pública, mas a URL carrega o
# token de uso único — nenhum robô tem o que fazer com ela (e visitá-la
# seria clicar no link no lugar do usuário). /verifique-email exige sessão.
Disallow: /confirmar-email
Disallow: /verifique-email

# URLs antigas, hoje apenas redirecionamentos para /co-chairs
Disallow: /dashboard
Disallow: /avaliadores
Disallow: /trabalhos
Disallow: /categorias
Disallow: /atribuicoes
Disallow: /rankings

# Público: / , /login , /cadastro , /cronograma
# O cronograma é conteúdo público de verdade (datas do congresso, sem
# dado de ninguém) e deve ser indexado — é o que um interessado procura
# antes de se cadastrar. Por isso NÃO entra na lista de Disallow acima,
# e nem em vercel.json / _headers como noindex.
# (/confirmar-email também é público, mas fica fora do índice — ver acima)
